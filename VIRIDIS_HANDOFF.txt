package com.viridis.mod.content;

import arc.math.geom.*;
import mindustry.game.*;
import mindustry.maps.generators.*;
import mindustry.world.*;

/** Процедурная генерация: перегной + мох + фризо-почва, стены ферро-флоры полосами шума. */
public class ViridisPlanetGenerator extends PlanetGenerator{
  @Override
  public void generate(Sector sector, PlanetParams plan){
    this.seed = sector.id * 7919 + baseSeed;
    super.generate(sector, plan);
  }

  @Override
  protected void genTile(Vec3 position, TileGen tile){
    tile.floor = ViridisBlocks.humus;
    float n = noise(position.x*0.6f, position.y*0.6f, 4d, 0.55d, 2.1d, (double)(seed & 0xFFFF));
    if(n > 0.62f) tile.floor = ViridisBlocks.sporeTurf;
    else if(n < 0.30f) tile.floor = ViridisBlocks.cryoSoil;
    if(noise(position.x*1.3f, position.y*1.3f, 3d, 0.5d, 2.4d, 991d) > 0.72f)
      tile.block = ViridisBlocks.ferroflora;
  }

  // ВНИМАНИЕ: сверить модификатор с родительским классом (PlanetGenerator.getSectorSize)
  // через zipfile перед записью; если protected — писать protected, если public — public.
  @Override
  public int getSectorSize(Sector sector){
    return sector.id == 17 ? 130 : 100 + (Math.abs(sector.id) % 5) * 15;
  }
}

def blob(path, sz, col, shape="circle"):
    im = Image.new("RGBA", (sz, sz), (0,0,0,0)); d = ImageDraw.Draw(im)
    m = max(1, sz//8)
    if shape=="circle": d.ellipse([m,m,sz-m,sz-m], fill=col+(255,))
    elif shape=="square": d.rectangle([m,m,sz-m,sz-m], fill=col+(255,))
    else: d.polygon([(sz//2,m),(sz-m,sz-m),(m,sz-m)], fill=col+(255,))
    d.rectangle([m,m,sz-1,sz-1], outline=(0,0,0,120))
    im.save(path)
